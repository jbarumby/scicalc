#!/usr/bin/env python3
"""NUMERIA ES8-P151: constructive multiscale density and exact moment algebra.

No claim of universal Weil positivity. Stdlib only; exact algebraic checks use
Fraction and integer arithmetic. Numerical C1 tests (optional) use mpmath.
"""
from __future__ import annotations
import argparse
import json
from fractions import Fraction


def level_step(m: int) -> Fraction:
    if m < 1:
        raise ValueError('m >= 1')
    return Fraction(1, 1 << m)


def grid_indices(N: int, m: int) -> range:
    """Centres j*2**(-m) in [-N,N]."""
    if N < 1 or m < 1 or m > N:
        raise ValueError('N >=1 and 1<=m<=N')
    J=N << m
    return range(-J, J+1)


def count_generators(N: int) -> int:
    """Raw generator count, including overlaps between scale levels.

    V_N is generated by g_(m,j) for 1<=m<=N, |j|<=N*2**m.
    """
    if N < 1:
        raise ValueError('N >= 1')
    return sum(2*(N << m)+1 for m in range(1,N+1))


def count_local_moment_basis(N: int) -> int:
    """Within each fixed scale, moment kernel dimension = n_m-2."""
    return sum(2*(N << m)-1 for m in range(1,N+1))


def approximate_error_bound(M1: Fraction, M2: Fraction, m: int) -> Fraction:
    """Exact rational bound for pre-projection C1 approximation error.

    M1 >= ||f'||inf, M2 >= ||f''||inf.
    ||f-P||_C1 <= d*M1 + (3d/2+d^2/8)*M2.
    """
    if M1<0 or M2<0:
        raise ValueError('nonnegative input bounds required')
    d=level_step(m)
    return d*M1 + (Fraction(3,2)*d+d*d/8)*M2


def conservative_projection_factor(A: int) -> int:
    """D_A <= 1+24*A*2**A for integer A>=2.

    Here |u|<=1, |u'|<=2, m_0>=1/2, sinh(1/2)>=1/2,
    exp(A/2)<=2**A. This deliberately overestimates operator norm.
    """
    if A < 2:
        raise ValueError('A >=2')
    return 1+24*A*(2**A)


def corrected_error_bound(M1: Fraction, M2: Fraction, m: int, A: int) -> Fraction:
    return conservative_projection_factor(A)*approximate_error_bound(M1,M2,m)


def single_scale_generator_count(m: int) -> int:
    """S_m = span of centres j*2^-m with |j*2^-m| <= m."""
    if m < 1:
        raise ValueError('m >= 1')
    return 2*m*(1 << m)+1


def single_scale_moment_dimension(m: int) -> int:
    return single_scale_generator_count(m)-2


def single_scale_projection_factor(A: int) -> int:
    """For m>=3: ||f - p_m^0||_C1 <= (1+48 A 2^A) E_m(f).

    A = max(ceil(L)+1,3). Uses translated, smoothed interpolations
    of two fixed correctors; the moment matrix is invertible exactly.
    """
    if A < 3:
        raise ValueError('A >= 3')
    return 1+48*A*(2**A)


def single_scale_admissible_error(M1: Fraction, M2: Fraction, m: int, A: int) -> Fraction:
    if m < 3:
        raise ValueError('single-scale estimate is established for m >=3')
    return single_scale_projection_factor(A)*approximate_error_bound(M1,M2,m)


def local_annihilator_laurent_zero(j: int) -> bool:
    """Verify q^(j-1) - (q+q^-1)q^j + q^(j+1) == 0,
    and the same expression with q -> q^-1, by Laurent polynomial coefficients.
    """
    for sign in (+1,-1):
        terms={}
        def term(exp,coef):
            terms[exp]=terms.get(exp,0)+coef
        term(sign*(j-1),1)
        term(sign*j+1,-1)
        term(sign*j-1,-1)
        term(sign*(j+1),1)
        if any(terms.values()):
            return False
    return True


def fixed_level_rank_and_kernel(m: int, N: int) -> dict:
    J=N<<m
    n=2*J+1
    # n linearly independent translates g_(m,j): left-edge support argument.
    # Two moment rows independent since q>1; local annihilator coefficient
    # vectors j=-J+1,...,J-1 independent by triangular first nonzero entry.
    return {'m':m,'N':N,'centres':n,'moment_rank':2,'kernel_dimension':n-2,
            'local_generators':n-2,
            'symbolic_moment_cancellation':all(local_annihilator_laurent_zero(j)
                                                 for j in range(-J+1,J))}


def exact_rational_toy_moment_projection():
    """Sanity check over Q(q) specialised at q=2 (not physical q=exp(d/2)).

    For 5 centres j=-2,-1,0,1,2, use common factor m_d=1,
    moments q^j and q^-j. Correct using centre j=+1 and j=-1.
    """
    q=Fraction(2)
    js=range(-2,3)
    pos=lambda j:q**j
    neg=lambda j:q**(-j)
    M=((pos(1),pos(-1)),(neg(1),neg(-1)))
    det=M[0][0]*M[1][1]-M[0][1]*M[1][0]
    assert det !=0
    results=[]
    for k in js:
        moments=(pos(k),neg(k))
        cu=(M[1][1]*moments[0]-M[0][1]*moments[1])/det
        cv=(-M[1][0]*moments[0]+M[0][0]*moments[1])/det
        residues=(moments[0]-cu*pos(1)-cv*pos(-1),
                  moments[1]-cu*neg(1)-cv*neg(-1))
        assert residues==(0,0)
        results.append({'j':k,'c_u':str(cu),'c_v':str(cv),'residues':['0','0']})
    return {'q':'2','det':str(det),'projections':results}


def report(N: int=8) -> dict:
    if N < 1:
        raise ValueError('N>=1')
    return {
        'stage':'ES8-P151',
        'status':'CONSTRUCTIVE_DENSITY_AND_EXACT_MOMENT_ALGEBRA_ONLY',
        'levels':[fixed_level_rank_and_kernel(m,N) for m in range(1,N+1)],
        'generator_count':count_generators(N),
        'single_scale_m':N,
        'single_scale_generators':single_scale_generator_count(N),
        'single_scale_moment_dimension':single_scale_moment_dimension(N),
        'local_moment_generators':count_local_moment_basis(N),
        'example_C1_error_bounds':[
            {'m':m,'d':str(level_step(m)),
             'E(M1=M2=1)':str(approximate_error_bound(Fraction(1),Fraction(1),m)),
             'D_A_for_A=3':conservative_projection_factor(3),
             'projected_error_bound':str(corrected_error_bound(Fraction(1),Fraction(1),m,3))}
            for m in range(1,min(N,8)+1)],
        'toy_moment_projection':exact_rational_toy_moment_projection(),
        'single_scale_explicit_errors': [
          {'m':m,'E_M1=M2=1':str(approximate_error_bound(Fraction(1),Fraction(1),m)),
           'D_A3':single_scale_projection_factor(3),
           'E_corrected_M1=M2=1':str(single_scale_admissible_error(Fraction(1),Fraction(1),m,3))}
          for m in range(3,min(N,12)+1)],
        'limitations':['No Gram matrix of the genuine Weil functional computed',
                       'No proof of universal semidefinite positivity',
                       'Smooth bump factor and finite-difference C1 error need conventional analytic lemmas',
                       'P111 finite prime-centre matrix is not a certificate for the enlarged family']}


def main():
    ap=argparse.ArgumentParser()
    ap.add_argument('--N',type=int,default=8)
    ap.add_argument('--out',default='P151_AUDIT.json')
    a=ap.parse_args()
    data=report(a.N)
    with open(a.out,'w',encoding='utf-8') as f:
        json.dump(data,f,ensure_ascii=False,indent=2)
        f.write('\n')
    print('N',a.N,'generators',data['generator_count'],'moment-basis-at-levels',data['local_moment_generators'])
    print('single-scale',data['single_scale_generators'],'admissible kernel dim',data['single_scale_moment_dimension'])
    print('exact Laurent cancellations',sum(x['local_generators'] for x in data['levels']))
    print('report',a.out)

if __name__=='__main__':
    main()
